\section{Reproducing the results}
\label{app:repro}
\begin{lstlisting}[language=bash, caption={From a clean checkout (Vivado/Vitis 2026.1, the card in a PCIe slot).}]
python3 setup.litex.du-pcb.py                      # LiteX 2026.08, venv, tool paths
git submodule update --init --recursive            # gemm_hls + hlslib (reference only)
sudo scripts/setup_permissions.sh                  # once: password-less driver and rescan

# one configuration: build, load, run
.venv/bin/python scripts/gemm_build.py configs/gemm/pe32x8_ta_t1024x512_225mhz.json
.venv/bin/python scripts/gemm_run.py build/gemm/soc/<build_id> --program \
        --sizes 4096x4096x4096 8192x8192x8192 --reps 2 --verify sampled

# a whole campaign (builds are reused across campaigns), then this report
.venv/bin/python scripts/gemm_dse.py configs/dse/pe32_ta_225_presets.json
.venv/bin/python scripts/gemm_dse.py configs/dse/int4_pe64.json     # int4 PE 64x64 (kernel.data_type/data_bits)
docs/report/build.sh
\end{lstlisting}

\paragraph{Where things are.}
\begin{itemize}
  \item \file{kernels/gemm/}: the adapted HLS kernel (\file{ORIGIN.md} lists every change against upstream)
  \item \file{hw/gemm/}: SoC, DMA bridge, kernel wrapper, AXI converters and counters, simulations
  \item \file{boards/dupcb/}: base SoC, DDR3 PHY and calibration, PCIe PHY fix
  \item \file{tcl/}: HLS flow, Vivado hooks, JTAG programming
  \item \file{sw/gemm\_host/}: C++ host library, \code{gemm\_runner}, \code{ddr\_dma\_test}
  \item \file{flow/gemmflow/} and \file{flow/tests/}: the Python flow and its unit tests
  \item \file{scripts/}: \code{gemm\_build.py}, \code{gemm\_run.py}, \code{gemm\_dse.py} and the board scripts
  \item \file{configs/gemm/}, \file{configs/dse/}: build configurations and campaigns
  \item \file{docs/notes1/}, \file{docs/notes2/}: plans and progress logs for float32 and for the integer and
        fixed-point formats (a dated record of every step); \file{docs/report/}: this report
  \item \file{misc/int32\_verify/}, \file{misc/pe\_forwarding/}: the DDR read-margin and tile-rule investigations
\end{itemize}
